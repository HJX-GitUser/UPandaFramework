using System;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEditor.Build.Reporting;
using UnityEngine;
public static class CoordinationPlayerBuild
{
    public static void Build()
    {
        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
        EditorSceneManager.SaveScene(UnityEngine.SceneManagement.SceneManager.GetActiveScene(),"Assets/CoordinationSmokeTarget.unity");
        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
        new GameObject("CoordinationSmoke").AddComponent<CoordinationPlayerSmoke>();
        EditorSceneManager.SaveScene(UnityEngine.SceneManagement.SceneManager.GetActiveScene(),"Assets/CoordinationSmoke.unity");
        PlayerSettings.productName="UPandaCoordinationVerification";PlayerSettings.companyName="UPandaAudit";
        var result=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/CoordinationSmoke.unity","Assets/CoordinationSmokeTarget.unity"},target=BuildTarget.StandaloneWindows64,locationPathName="CoordinationBuild/CoordinationSmoke.exe",options=BuildOptions.Development});
        if(result.summary.result!=BuildResult.Succeeded)throw new Exception("Coordination player build failed");
        Debug.Log("COORDINATION_PLAYER_BUILD_SUCCEEDED");
    }
}
